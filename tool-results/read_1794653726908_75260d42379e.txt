    21→def detect_country_code(timeout: float = 4.0):
    22→    """يرجع رمز البلد فقط (مثل BD / SA / EG) عبر IP العام. لا يجلب أي موقع دقيق."""
    23→    if _GEO_CACHE["country"] and time.time() - _GEO_CACHE["checked_at"] < 3600:
    24→        return _GEO_CACHE["country"]
    25→    endpoints = [
    26→        ("http://ip-api.com/json/?fields=status,countryCode", lambda d: (d or {}).get("countryCode")),
    27→        ("https://ipapi.co/json/", lambda d: (d or {}).get("country_code")),
    28→        ("https://api.ip.sb/geoip", lambda d: (d or {}).get("country_code")),
    29→    ]
    30→    for url, extract in endpoints:
    31→        try:
    32→            r = requests.get(url, timeout=timeout, headers={"User-Agent": USER_AGENT})
    33→            code = extract(r.json())
    34→            if code:
    35→                _GEO_CACHE["country"] = str(code).upper()
    36→                _GEO_CACHE["checked_at"] = time.time()
    37→                return _GEO_CACHE["country"]
    38→        except Exception:
    39→            continue
    40→    return None

   332→def _format_raw_candles(raw):
   333→    """تنسيق رد الخادم الرسمي: تحليل _parse_raw_candles الأصلي + محاذاة
   334→    _align_time الأصلية + إزالة تكرار وترتيب — نفس حلقة التنسيق في
   335→    fetch_candles_once الأصلية."""
   336→    if not raw:
   337→        return []
   338→    parsed = _parse_raw_candles(raw)          # الأصل حرفياً
   339→    formatted = []
   340→    for c in parsed:
   341→        try:
   342→            aligned = _align_time(int(c["time"]))
   343→            formatted.append({"time": aligned, "open": c["open"], "high": c["high"],
   344→                              "low": c["low"], "close": c["close"],
   345→                              "volume": int(c.get("volume", 0) or 0)})
   346→        except Exception:
   347→            continue
   348→    seen, unique = set(), []
   349→    for c in formatted:
   350→        if c["time"] not in seen:
   351→            seen.add(c["time"])
   352→            unique.append(c)
   353→    unique.sort(key=lambda x: x["time"])
   354→    return unique

   356→async def qx_fetch_window_raw(api, end_ts, seconds, tf_sec, foreground=False, timeout=45.0):
   357→    """بوابة الجلب الأمامية الواحدة — نداء CLIENT.get_candles الأصلي نفسه
   358→    (المُجرّب والآمن). foreground يمنع اختلاط طلب العرض الحالي مع التحميل المسبق."""
   359→    global _QX_WINDOW_BUSY
   360→    if CLIENT is None or CLIENT.api is None:
   361→        return []
   362→    if foreground:
   363→        for _ in range(int(max(1.0, float(timeout) * 10))):
   364→            if not _QX_WINDOW_BUSY:
   365→                break
   366→            await asyncio.sleep(0.1)
   367→    _QX_WINDOW_BUSY = True
   368→    try:
   369→        res = await asyncio.wait_for(
   370→            CLIENT.get_candles(api, int(end_ts), int(max(60, seconds)), int(tf_sec)),
   371→            timeout=float(timeout))
   372→        return list(res or [])
   373→    except Exception:
   374→        return []
   375→    finally:
   376→        _QX_WINDOW_BUSY = False

   378→async def fetch_older_batch(api, before_ts, batch=300):
   379→    """دفعة أقدم للتمرير للخلف — نداء CLIENT.get_candles الأصلي بمرساة before_ts،
   380→    والدمج في ذاكرة الأصل بلا تقليص (setdefault: القديم الناقص فقط يُضاف)."""
   381→    if CLIENT is None or CLIENT.api is None:
   382→        return 0
   383→    psec = max(1, int(PERIOD_SECONDS))
   384→    batch = max(50, int(batch or 300))
   385→    try:
   386→        res = await asyncio.wait_for(
   387→            CLIENT.get_candles(api, int(before_ts), batch * psec, psec), timeout=45.0)
   388→    except Exception:
   389→        return 0
   390→    if not res:
   391→        return 0
   392→    candles = [c for c in res
   393→               if isinstance(c, dict) and int(c.get("time", 0) or 0) < int(before_ts)]
   394→    if not candles:
   395→        return 0
   396→    asset = _get_or_create_asset(api)
   397→    if asset is None:
   398→        return 0
   399→    merged_map = {c["time"]: c for c in asset.candles}
   400→    for c in candles:
   401→        merged_map.setdefault(c["time"], c)
   402→    merged = sorted(merged_map.values(), key=lambda x: x["time"])
   403→    asset.candles = merged[-int(ASSET_MAX_CANDLES):]
   404→    if asset.candles:
   405→        asset.price = asset.candles[-1]["close"]
   406→    asset.mark_dirty()
   407→    return len(candles)

   412→    CLIENT._fetch_historical_batch الأصلية، نفس chunk=period×FETCH_CHUNK_SIZE،

   414→    FETCH_BATCH_DELAY) — الفرق الوحيد: كل دفعة تُسلَّم فور وصولها عبر
   415→    batch_callback(batch, worker_id) ليرفعها التطبيق إلى الشارت حية."""
   416→    if CLIENT is None or CLIENT.api is None:
   417→        return []
   418→    max_workers = max_workers or 1
   419→    chunk_seconds = period * FETCH_CHUNK_SIZE
   420→    all_candles = {}
   421→    current_time = int(time.time())
   422→    target_start_time = current_time - amount_of_seconds
   423→    block_size = max(1, amount_of_seconds // max_workers)
   424→    semaphore = asyncio.Semaphore(max_workers)
   425→
   426→    async def worker(start_t, end_t, worker_id):
   427→        worker_candles = {}
   428→        async with semaphore:
   429→            oldest_t = start_t
   430→            consecutive_failures_in_worker = 0
   431→            while oldest_t > end_t:
   432→                if not CLIENT or not getattr(getattr(CLIENT, "api", None), "state", None) \
   433→                        or not getattr(CLIENT.api.state, "check_accepted_connection", False):
   434→                    break
   435→                index = next(_request_counter)
   436→                batch_data = await CLIENT._fetch_historical_batch(
   437→                    asset, oldest_t, chunk_seconds, period, index, timeout)
   438→                if not batch_data:
   439→                    oldest_t -= chunk_seconds
   440→                    consecutive_failures_in_worker += 1
   441→                    if consecutive_failures_in_worker >= 3:
   442→                        break
   443→                    await asyncio.sleep(FETCH_BATCH_DELAY * 2)
   444→                    continue
   445→                consecutive_failures_in_worker = 0
   446→                new_batch = CLIENT._parse_historical_candles(batch_data)
   447→                if not new_batch:
   448→                    oldest_t -= chunk_seconds
   449→                    continue
   450→                batch_times = []
   451→                for c in new_batch:
   452→                    ts = c["time"]
   453→                    if ts >= end_t and ts <= start_t:
   454→                        worker_candles[ts] = c
   455→                        batch_times.append(ts)
   456→                if not batch_times:
   457→                    oldest_t -= chunk_seconds
   458→                    continue
   459→                batch_times.sort()
   460→                new_oldest = batch_times[0]
   461→                if batch_callback:
   462→                    try:
   463→                        batch_callback([worker_candles[t] for t in batch_times], worker_id)
   464→                    except Exception:
   465→                        pass
   466→                oldest_t = new_oldest if new_oldest < oldest_t else oldest_t - chunk_seconds
   467→                await asyncio.sleep(FETCH_BATCH_DELAY)
   468→        return list(worker_candles.values())
   469→
   470→    try:
   471→        await CLIENT.start_candles_stream(asset, period)
   472→    except Exception:
   473→        pass
   474→    tasks = []
   475→    for i in range(max_workers):
   476→        s = current_time - (i * block_size)
   477→        e = max(target_start_time, s - block_size)
   478→        tasks.append(worker(s, e, i))
   479→    results = await asyncio.gather(*tasks)
   480→    for batch in results:
   481→        for c in batch:
   482→            all_candles[c["time"]] = c
   483→    return sorted(all_candles.values(), key=lambda x: x["time"])

   485→async def v14_deep_gap_seal(api, reason=""):
   486→    """السد العميق: مرساة عند أقدم شمعة اصطناعية (volume==0) — نداء الجلب
   487→    الأصلي CLIENT.get_candles يعيد شموعاً حقيقية تُستبدل بالشرطات المسطحة داخل
   488→    نافذة محدودة (حتى 200 شمعة) دون لمس الشمعة الحية ولا تقليص الذاكرة."""
   489→    if CLIENT is None or CLIENT.api is None:
   490→        return 0
   491→    asset = _get_or_create_asset(api)
   492→    if asset is None or len(asset.candles) < 2:
   493→        return 0
   494→    psec = max(1, int(PERIOD_SECONDS))
   495→    cur_start = _align_time(int(time.time()))
   496→    art = [c for c in asset.candles
   497→           if int(c.get("volume", 1) or 0) == 0 and c["time"] < cur_start]
   498→    if not art:
   499→        return 0
   500→    oldest = min(c["time"] for c in art)
   501→    span = max(1, (cur_start - oldest) // psec)
   502→    window = min(200, span + 10)
   503→    try:
   504→        res = await asyncio.wait_for(
   505→            CLIENT.get_candles(api, cur_start + psec, window * psec, psec),
   506→            timeout=30.0)
   507→    except Exception:
   508→        return 0
   509→    if not res:
   510→        return 0
   511→    real = {}
   512→    for c in res:
   513→        if isinstance(c, dict) and int(c.get("time", 0) or 0) > 0:
   514→            real[int(c["time"])] = c
   515→    if not real:
   516→        return 0
   517→    merged = []
   518→    for c in asset.candles:
   519→        if c["time"] >= cur_start:
   520→            merged.append(c)                      # الشمعة الحية لا تُلمس
   521→            continue
   522→        r = real.get(c["time"])
   523→        if r is not None and int(c.get("volume", 1) or 0) == 0:
   524→            merged.append(dict(r))
   525→        else:
   526→            merged.append(c)
   527→    have = {c["time"] for c in merged}
   528→    for t, r in real.items():
   529→        if t not in have and t < cur_start:
   530→            merged.append(dict(r))
   531→    merged.sort(key=lambda x: x["time"])
   532→    asset.candles = merged[-int(ASSET_MAX_CANDLES):]
   533→    if asset.candles:
   534→        asset.price = asset.candles[-1]["close"]
   535→    asset.mark_dirty()
   536→    V14_DEEP_REPAIRED[api] = time.time()
   537→    return len(real)

   544→async def tick_cache_collector():
   545→    """يلتقط كل تيك يصل (realtime_candles الأصلية — يتحدث عند كل تيك) في ذاكرة
   546→    كل عملة بث حي، بلا أي كتابة على القرص. يبني منه التطبيق فريمات الثواني."""
   547→    while True:
   548→        try:
   549→            if CLIENT is not None and getattr(CLIENT, "api", None) is not None:
   550→                rc = getattr(CLIENT.api, "realtime_candles", {}) or {}
   551→                for sym, tick in list(rc.items()):
   552→                    if not isinstance(tick, (list, tuple)) or len(tick) < 3:
   553→                        continue
   554→                    try:
   555→                        ts, price = int(tick[1]), float(tick[2])
   556→                    except Exception:
   557→                        continue
   558→                    if price <= 0 or ts <= 0:
   559→                        continue
   560→                    q = _TICK_CACHE.get(sym)
   561→                    if q is None:
   562→                        q = _qx_deque(maxlen=_TICK_QUEUE_MAX)
   563→                        _TICK_CACHE[sym] = q
   564→                    if q and q[-1][0] == ts and q[-1][1] == price:
   565→                        continue
   566→                    q.append((ts, price))
   567→        except Exception:
   568→            pass
   569→        await asyncio.sleep(0.05)

   571→async def tick_retention_loop():
   572→    """حذف التيكات الأقدم من 5 دقائق (ذاكرة فقط)."""
   573→    while True:
   574→        try:
   575→            now = time.time()
   576→            for sym, q in list(_TICK_CACHE.items()):
   577→                while q and (now - q[0][0]) > _TICK_MAX_AGE:
   578→                    q.popleft()
   579→                if not q:
   580→                    _TICK_CACHE.pop(sym, None)
   581→        except Exception:
   582→            pass
   583→        await asyncio.sleep(30.0)

   652→# ═══ 9) التحميل المسبق المتسلسل — «ضع كل العملات واحدة تلو الأخرى» ═══
   653→async def preload_all_assets():
   654→    """التحميل المسبق المتسلسل لكل العملات المفتوحة **واحدة تلو الأخرى** —
   655→    نفس تدفق الكتلة الرئيسية للأصل حرفياً لكل عملة:
   656→      realtime_stream أولاً (يجمع التيك أثناء الجلب) ← fetch_candles_with_retry
   657→      الأصلية (تاريخ FETCH_DURATION_SECONDS + الجسر + سد الفجوات + HST) ←
   658→      فاصل بين عملة وأخرى.
   659→    فاتحة الشارت أولاً (BRLUSD_otc) ثم بقية العملات المفتوحة من instruments
   660→    بالترتيب. كل عملة تُسجَّل في _PRELOADED_INFO لحالة التحميل في الواجهة."""
   661→    if CLIENT is None or CLIENT.api is None:
   662→        return 0
   663→    # اطلب قائمة instruments مرة واحدة (الطريقة الرسمية) وانتظر وصولها باختصار
   664→    try:
   665→        await fetch_instruments_now(timeout=12.0)
   666→    except Exception:
   667→        pass
   668→    try:
   669→        await wait_until(lambda: bool(_instruments_snapshot()),
   670→                         timeout=15.0, poll_interval=0.5)
   671→    except asyncio.TimeoutError:
   672→        pass
   673→    symbols, seen = [], set()
   674→    opener = PRELOAD_FIRST_SYMBOL
   675→    if opener:
   676→        symbols.append(opener)
   677→        seen.add(opener)
   678→    entries = _instruments_snapshot()
   679→    for e in entries:
   680→        sym = str((e or {}).get("asset") or "").strip()
   681→        if not sym or sym in seen:
   682→            continue
   683→        if e.get("open") is False:
   684→            continue        # المغلقة الآن لا بث حي لها — تُحمّل عند فتح شارتها
   685→        symbols.append(sym)
   686→        seen.add(sym)
   687→    if len(symbols) <= 1:
   688→        # fallback: قائمة الأصل الافتراضية ASSET_LIST كما هي
   689→        for sym in list(ASSET_LIST):
   690→            if sym not in seen:
   691→                symbols.append(sym)
   692→                seen.add(sym)
   693→    total = len(symbols)
   694→    logmsg(f"[preload] sequential preload: {total} assets, ONE after another "
   695→           f"(history window {FETCH_DURATION_SECONDS}s each)")
   696→    loaded = 0
   697→    streak = 0
   698→    for idx, sym in enumerate(symbols, 1):
   699→        if not CONNECTION_ALIVE or CLIENT is None or CLIENT.api is None:
   700→            logmsg(f"[preload] connection lost at {idx}/{total}; waiting reconnect…")
   701→            try:
   702→                await wait_until(
   703→                    lambda: CONNECTION_ALIVE and CLIENT is not None and CLIENT.api is not None,
   704→                    timeout=30.0, poll_interval=0.5)
   705→            except asyncio.TimeoutError:
   706→                streak += 1
   707→                if streak >= MAX_RECONNECTS_DURING_FETCH:
   708→                    logmsg("[preload] giving up after repeated disconnects")
   709→                    break
   710→                continue
   711→        streak = 0
   712→        asset = _get_or_create_asset(sym)
   713→        if asset is None:
   714→            continue
   715→        try:
   716→            st = getattr(asset, "stream_task", None)
   717→            if st is None or (hasattr(st, "done") and st.done()):
   718→                asset.streaming = False
   719→                asset.stream_task = asyncio.ensure_future(realtime_stream(asset))
   720→        except Exception:
   721→            pass
   722→        if len(asset.candles) < MIN_CANDLES_THRESHOLD:
   723→            try:
   724→                n, _att = await fetch_candles_with_retry(
   725→                    asset, max_retries=MAX_FETCH_RETRIES, idx=idx, total=total)
   726→                if n >= MIN_CANDLES_THRESHOLD:
   727→                    loaded += 1
   728→            except Exception as e:
   729→                logmsg(f"[preload] {sym}: {str(e)[:80]}")
   730→        else:
   731→            loaded += 1
   732→        _PRELOADED_INFO[sym] = {"candles": len(asset.candles), "at": time.time()}
   733→        if idx % 10 == 0:
   734→            logmsg(f"[preload] progress: {idx}/{total} assets ({loaded} loaded)")
   735→        # فاصل بين عملة وأخرى — نفس الأصل: واحدة تلو الأخرى بلا تسرّع
   736→        await asyncio.sleep(1.0 if PERIOD_SECONDS < 60 else FETCH_ASSET_DELAY)
   737→    logmsg(f"[preload] done: {loaded}/{total} assets loaded sequentially")
   738→    return loaded
