  3517→# CandleStore: SQLite-backed persistent candle storage.
  3518→# ------------------------------------------------------------------------------
  3519→# Why SQLite instead of JSON files:
  3520→#   - JSON is slow: full file rewrite on every save, full file load on every
  3521→#     read, no indexing. For ~1000 candles per asset across many assets this
  3522→#     wastes I/O and risks corruption if the process is killed mid-write.
  3523→#   - SQLite is transactional, atomic, indexed by (symbol, time), and
  3524→#     supports efficient range queries ("give me all candles for symbol X
  3525→#     from time A to time B") without loading the whole table into memory.
  3526→#   - Concurrent reads + a single writer are safe out of the box.
  3527→#
  3528→# Schema:
  3529→#   candles(symbol TEXT, time INTEGER, open REAL, high REAL, low REAL,
  3530→#           close REAL, volume INTEGER, PRIMARY KEY(symbol, time))
  3531→#   gap_state(symbol TEXT PRIMARY KEY, filled INTEGER, last_fill_at REAL)
  3532→#       - 'filled' is 0/1 and tracks whether fill_gap_once has already
  3533→#         succeeded for this asset. Once it is 1, MT4Writer.fill_gap_once
  3534→#         stops touching closed candles for this asset entirely.
  3535→# ------------------------------------------------------------------------------
  3536→import sqlite3 as _sqlite3
  3537→import threading as _threading

  3891→    def generate_hst_buffer(self, symbol: str, period: int, candles: list) -> bytes:
  3892→        # For sub-minute timeframes, write period=1 (M1) in the HST header
  3893→        # so MT4 opens it as a normal M1 file. The candles inside have actual
  3894→        # 5s/10s/15s/30s timestamps. For 60s+ timeframes, store the MT4 period int.

  3897→        struct.pack_into('<I', header, 0, 400)
  3898→        struct.pack_into('64s', header, 4, b'MetaQuotes Software Corp.'.ljust(64, b'\x00'))
  3899→        struct.pack_into('12s', header, 68, symbol[:12].encode('ascii').ljust(12, b'\x00'))
  3900→        struct.pack_into('<I', header, 80, header_period)
  3901→        first_price = candles[0]['close'] if candles else 0.0
  3902→        digits = self.get_digits(symbol, first_price)
  3903→        struct.pack_into('<I', header, 84, digits)
  3904→        struct.pack_into('<I', header, 88, 0)

  3909→        offset = 148

  3912→            struct.pack_into('<d', buffer, offset + 4, float(c['open']))
  3913→            struct.pack_into('<d', buffer, offset + 12, float(c['low']))
  3914→            struct.pack_into('<d', buffer, offset + 20, float(c['high']))
  3915→            struct.pack_into('<d', buffer, offset + 28, float(c['close']))
  3916→            struct.pack_into('<Q', buffer, offset + 36, int(c.get('volume', 0)))
  3917→            offset += 44

  3934→        aligned_time = _align_time(tick['time'])
  3935→
  3936→        # FIX STALE-CANDLE v2: only the last candle (current/open) can be updated.
  3937→        # All candles before it are closed and must never be touched.
  3938→        if state['candles']:

  3946→                last['high'] = max(last['high'], tick['price'])
  3947→                last['low'] = min(last['low'], tick['price'])
  3948→                last['close'] = tick['price']
  3949→                last['volume'] += 1
  3950→            else:

  3952→                period_secs = PERIOD_SECONDS if PERIOD_SECONDS > 0 else 60

  3956→                    last_close = state['candles'][-1]['close']

  3960→                            'time': fill_time,
  3961→                            'open': last_close,
  3962→                            'high': last_close,
  3963→                            'low': last_close,
  3964→                            'close': last_close,
  3965→                            'volume': 0,
  3966→                        })

  3969→                state['candles'].append({
  3970→                    'time': aligned_time, 'open': tick['price'], 'high': tick['price'],

  3972→                })
  3973→        else:

  3977→            })

  4150→    async def fill_gap_once(self, client, asset_obj, period: int):

  4156→ 3. _fill_missing_slots => .
  4157→ 4. 0 (volume=0) => .

  4163→    every subsequent call to fill_gap_once returns immediately without

  4177→        # (gap_state.filled == 1 in the SQLite store), do NOT touch the candles
  4178→        # at all. This is the entire fix for the 'candles flicker' symptom -
  4179→        # the original algorithm is preserved 1:1, we only skip it after the
  4180→        # first successful run.
  4181→        try:
