  1832→# Fill missing slots with flat candles. max_gap=1440 (full day) allows filling
  1833→# gaps up to 1 day. period_seconds=None reads PERIOD_SECONDS dynamically.
  1834→def _fill_missing_slots(candles: list, period_seconds: int = None,
  1835→                        max_gap: int = 1440) -> list:
  1836→    """Fills time gaps between candles with flat candles (previous close).
  1837→
  1838→ - Added candles have volume=0 (marked synthetic).
  1839→ - max_gap: max number of periods to fill (1440 = 1 day for M1).
  1840→ - period_seconds=None reads PERIOD_SECONDS dynamically (multi-timeframe).
  1841→ """
  1842→    if period_seconds is None:
  1843→        period_seconds = PERIOD_SECONDS
  1844→    if len(candles) < 2:
  1845→        return list(candles)
  1846→    candles = sorted(candles, key=lambda x: x['time'])
  1847→    out = [dict(candles[0])]
  1848→    for i in range(1, len(candles)):
  1849→        prev = out[-1]
  1850→        curr = candles[i]
  1851→        gap = (curr['time'] - prev['time']) // period_seconds
  1852→        if 0 < gap <= max_gap:
  1853→
  1854→            slot = prev['time'] + period_seconds
  1855→            while slot < curr['time']:
  1856→                out.append({
  1857→                    'time': slot,
  1858→                    'open': prev['close'],
  1859→                    'high': prev['close'],
  1860→                    'low': prev['close'],
  1861→                    'close': prev['close'],
  1862→                    'volume': 0,
  1863→                })
  1864→                slot += period_seconds
  1865→        out.append(dict(curr))
  1866→    return out

  1868→# Aggregate 1-second candles into target_period (5s, 10s, 15s, 30s) candles.
  1869→# Used for sub-60s timeframes: fetch period=1 from server, aggregate locally.
  1870→def _aggregate_sub_minute(candles: list, target_period: int) -> list:
  1871→    """Aggregate 1-second candles into target_period candles.
  1872→
  1873→    For each group of (target_period) consecutive 1s candles:
  1874→    - time = first candle's time aligned to target_period boundary
  1875→    - open = first candle's open
  1876→    - high = max of all highs
  1877→    - low = min of all lows
  1878→    - close = last candle's close
  1879→    - volume = sum of all volumes
  1880→    """
  1881→    if not candles or target_period <= 1:
  1882→        return list(candles)
  1883→    candles = sorted(candles, key=lambda x: x['time'])
  1884→    buckets = OrderedDict()
  1885→    for c in candles:
  1886→        bucket_time = (c['time'] // target_period) * target_period
  1887→        if bucket_time not in buckets:
  1888→            buckets[bucket_time] = {
  1889→                'time': bucket_time,
  1890→                'open': c['open'],
  1891→                'high': c['high'],
  1892→                'low': c['low'],
  1893→                'close': c['close'],
  1894→                'volume': int(c.get('volume', 0) or 0),
  1895→            }
  1896→        else:
  1897→            b = buckets[bucket_time]
  1898→            b['high'] = max(b['high'], c['high'])
  1899→            b['low'] = min(b['low'], c['low'])
  1900→            b['close'] = c['close']
  1901→            b['volume'] += int(c.get('volume', 0) or 0)
  1902→    return list(buckets.values())

  2442→            # تتبع الـ chunks الفارغة المتتالية (للإيقاف المبكر الذكي)
  2443→            empty_chunks_in_a_row = 0
  2444→            while len(all_candles_1s) < target_1s_count and chunks_done < max_chunks:
  2445→                chunk_end = current_end - (chunks_done * chunk_seconds)
  2446→                try:
  2447→                    res_chunk = await asyncio.wait_for(
  2448→                        CLIENT.get_candles(api_name, chunk_end, chunk_seconds, fetch_period),
  2449→                        timeout=20,
  2450→                    )
  2451→                except Exception as e:
  2452→                    if chunks_done == 0:
  2453→                        logmsg(f"[fetch:{display_name}] get_candles(period=1) chunk {chunks_done+1} failed: {e}")
  2454→                    # retry once on transient error
  2455→                    if empty_chunks_in_a_row < 2:
  2456→                        empty_chunks_in_a_row += 1
  2457→                        await asyncio.sleep(0.2)
  2458→                        continue
  2459→                    break
  2460→                if not res_chunk or len(res_chunk) == 0:
  2461→                    empty_chunks_in_a_row += 1
  2462→                    # 3 chunks فارغة متتالية = نهاية التاريخ المتاح → توقف
  2463→                    if empty_chunks_in_a_row >= 3:
  2464→                        break
  2465→                    chunks_done += 1
  2466→                    continue
  2467→                # عتبة إيقاف أقل صرامة (من 50 → 10): بعض الأصول تُرجع 20-30 شمعة فقط في نهاية الأسبوع
  2468→                # هذا طبيعي ولا يعني أننا وصلنا لنهاية التاريخ
  2469→                if len(res_chunk) < 10 and chunks_done > 0:
  2470→                    all_candles_1s.extend(res_chunk)
  2471→                    break
  2472→                empty_chunks_in_a_row = 0
  2473→                all_candles_1s.extend(res_chunk)
  2474→                chunks_done += 1
  2475→                try:
  2476→                    pct = min(len(all_candles_1s) / target_1s_count, 1.0)
  2477→                    on_progress(pct * FETCH_DURATION_SECONDS, FETCH_DURATION_SECONDS)
  2478→                except Exception:
  2479→                    pass
  2480→                # delay صغير (0.03 بدل 0.05) لتسريع الجلب
  2481→                await asyncio.sleep(0.03)
  2482→            # De-duplicate by time
  2483→            seen_ts = set()
  2484→            unique_1s = []
  2485→            for c in all_candles_1s:
  2486→                if isinstance(c, dict):
  2487→                    ts = int(float(c.get('time', c.get('timestamp', 0))))
  2488→                    if ts not in seen_ts:
  2489→                        seen_ts.add(ts)
  2490→                        unique_1s.append(c)
  2491→            candles = unique_1s
  2492→            # No log message here — fetch_candles_with_retry prints the final count
  2493→        except Exception as e:
  2494→            print()
  2495→            logmsg(f"[fetch:{display_name}] get_candles(period=1) failed: {e}")
  2496→    else:
  2497→        # 60s+ timeframe — use parallel get_historical_candles
  2498→        if hasattr(CLIENT, 'get_historical_candles'):
  2499→            try:
  2500→                on_progress = _make_progress_printer(display_name, time.time(), idx, total)
  2501→                res = await asyncio.wait_for(
  2502→                    CLIENT.get_historical_candles(
  2503→                        api_name,
  2504→                        amount_of_seconds=FETCH_DURATION_SECONDS,
  2505→                        period=PERIOD_SECONDS,
  2506→                        max_workers=FETCH_MAX_WORKERS,
  2507→                        progress_callback=on_progress
  2508→                    ),
  2509→                    timeout=60,
  2510→                )
  2511→                if res and len(res) > 0: candles = res
  2512→            except Exception as e:
  2513→                print()
  2514→                logmsg(f"[fetch:{display_name}] get_historical_candles failed: {e}")
  2515→    if candles:
  2516→        formatted = []
  2517→        for c in candles:
  2518→            if not isinstance(c, dict): continue
  2519→            try:
  2520→                ts = int(float(c.get("time", c.get("timestamp", 0))))
  2521→                # Align to fetch_period (1s for sub-minute, PERIOD_SECONDS otherwise)
  2522→                aligned = _align_time(ts, period_seconds=fetch_period)
  2523→                o, h, l, cl = float(c.get("open", 0)), float(c.get("high", c.get("max", 0))), float(c.get("low", c.get("min", 0))), float(c.get("close", 0))
  2524→                if o > 0 and h > 0 and l > 0 and cl > 0:
  2525→                    formatted.append({'time': aligned, 'open': o, 'high': h, 'low': l, 'close': cl, 'volume': random.randint(50, 200)})
  2526→            except Exception: continue
  2527→        # Aggregate 1s candles into target subsecond timeframe
  2528→        if is_sub_minute and formatted:
  2529→            formatted = _aggregate_sub_minute(formatted, PERIOD_SECONDS)
  2530→        seen, unique = set(), []
  2531→        for c in formatted:
  2532→            if c['time'] not in seen:
  2533→                seen.add(c['time'])
  2534→                unique.append(c)
  2535→        unique.sort(key=lambda x: x['time'])
  2536→        return unique[-INITIAL_CANDLES:]
  2537→    return []

  2541→# ===== BRIDGE FETCH HELPER (سد فجوة الشموع الأخيرة بين التاريخي والبث) =====

  2555→async def _bridge_fetch_recent_candles(asset: Asset, existing_candles: list) -> list:

  2557→

  2559→        asset: كائن Asset
  2560→        existing_candles: قائمة الشموع التاريخية المُجلبَة مسبقاً

  2574→        return existing_candles
  2575→    if not getattr(CLIENT.api.state, 'check_accepted_connection', False):

  2586→

  2588→        async with _BRIDGE_FETCH_SEMAPHORE:
  2589→            is_sub_minute = PERIOD_SECONDS < 60
  2590→            # ✨ السر: استخدم time.time() الحالي (وليس T0 عند بداية الجلب التاريخي)

  2592→            bridge_candles = []

  2596→                # هامش 5x لضمان تغطية كاملة للفجوة الزمنية الكبيرة المحتملة

  2600→                try:

  2602→                        CLIENT.get_candles(
  2603→                            asset.api_symbol,
  2604→                            bridge_end_time,

  2608→                        timeout=BRIDGE_FETCH_TIMEOUT,
  2609→                    )
  2610→                except asyncio.TimeoutError:
  2611→                    logmsg(f"[bridge:{asset.symbol}] 1s fetch timeout (non-fatal)")
  2612→                    return existing_candles
  2613→                except Exception as e:
  2614→                    logmsg(f"[bridge:{asset.symbol}] 1s fetch failed (non-fatal): {e}")
  2615→                    return existing_candles
  2616→
  2617→                if not bridge_1s:
  2618→                    return existing_candles
  2619→
  2620→                # تنسيق شموع 1s

  2629→                        h = float(c.get("high", c.get("max", 0)) or 0)
  2630→                        l = float(c.get("low", c.get("min", 0)) or 0)
  2631→                        cl = float(c.get("close", c.get("c", 0)) or 0)

  2634→                                'time': aligned, 'open': o, 'high': h,
  2635→                                'low': l, 'close': cl, 'volume': 1

  2641→                bridge_candles = _aggregate_sub_minute(bridge_formatted, PERIOD_SECONDS)

  2689→                        o = float(c.get("open", 0) or 0)
  2690→                        h = float(c.get("high", 0) or 0)
  2691→                        l = float(c.get("low", 0) or 0)
  2692→                        cl = float(c.get("close", 0) or 0)
  2693→                        if o > 0 and h > 0 and l > 0 and cl > 0:

  2695→                                'time': aligned, 'open': o, 'high': h,
  2696→                                'low': l, 'close': cl, 'volume': 1

  2698→                    except Exception:
  2699→                        continue

  2701→            if not bridge_candles:
  2702→                return existing_candles
  2703→
  2704→            # ── الدمج الآمن: الجسر يستبدل التاريخي في نفس الفترة الزمنية ───

  2744→async def fetch_candles_with_retry(asset: Asset, max_retries=3, idx=1, total=1):
  2745→    # For sub-minute, use lower threshold + 1 retry (server caps ~500-1000 1s candles)
  2746→    is_sub_minute = PERIOD_SECONDS < 60
  2747→    threshold = 20 if is_sub_minute else MIN_CANDLES_THRESHOLD
  2748→    max_retries = 1 if is_sub_minute else MAX_FETCH_RETRIES
  2749→    last_error = None
  2750→    for attempt in range(1, max_retries + 1):
  2751→
  2752→        if not CONNECTION_ALIVE or CLIENT is None or CLIENT.api is None:
  2753→            logmsg(f"[fetch_retry:{asset.symbol}] connection dead before attempt {attempt}; waiting for reconnect...")
  2754→            # (timeout 15s)
  2755→            try:
  2756→                await wait_until(
  2757→                    lambda: CONNECTION_ALIVE and CLIENT is not None and CLIENT.api is not None,
  2758→                    timeout=15.0,
  2759→                    poll_interval=0.5,
  2760→                )
  2761→            except asyncio.TimeoutError:
  2762→                logmsg(f"[fetch_retry:{asset.symbol}] reconnect did not happen within 15s; giving up on this asset")
  2763→                return 0, attempt
  2764→        try:
  2765→            candles = await fetch_candles_once(asset, idx, total)
  2766→        except Exception as e:
  2767→            candles = []
  2768→            last_error = e
  2769→            logmsg(f"[fetch_retry:{asset.symbol}] attempt {attempt}/{max_retries} raised: {e}")
  2770→        # ===== BRIDGE FETCH: سد فجوة 4-5 شموع بين التاريخي والبث المباشر =====
  2771→        # استدعاء آمن: أي خطأ يُرجع candles كما هي دون كسر العملية.
  2772→        # يُجرّب آخر BRIDGE_FETCH_CANDLES شمعة تنتهي عند time.time() الحالي.
  2773→        if candles and len(candles) > 0:
  2774→            try:
  2775→                candles = await _bridge_fetch_recent_candles(asset, candles)
  2776→            except Exception as bridge_err:
  2777→                # الأمان التام: لا تُكسر الجلب التاريخي أبداً
  2778→                logmsg(f"[fetch_retry:{asset.symbol}] bridge skipped (non-fatal): {bridge_err}")
  2779→        if len(candles) >= threshold:
  2780→            # Merge historical candles with any live ticks collected during fetch
  2781→            # (realtime_stream may have appended live candles to asset.candles)
  2782→            live_candles = []
  2783→            if asset.candles:
  2784→                hist_times = set(c['time'] for c in candles)
  2785→                for c in asset.candles:
  2786→                    if c['time'] not in hist_times and c['time'] > (candles[-1]['time'] if candles else 0):
  2787→                        live_candles.append(c)
  2788→            asset.candles = candles + live_candles
  2789→            if asset.candles: asset.price = asset.candles[-1]['close']
  2790→            asset.write(force=True)
  2791→            # Update MT4_WRITER.sync_state with the full historical + live candles
  2792→            # so update_candle doesn't overwrite HST with only the few live candles
  2793→            MT4_WRITER.sync_state[asset.symbol] = {
  2794→                'period': PERIOD,
  2795→                'candles': list(asset.candles),
  2796→                'last_write': 0,
  2797→                'last_price': float(asset.candles[-1]['close']) if asset.candles else 0.0,
  2798→            }
  2799→            return len(asset.candles), attempt
  2800→
  2801→        # Merge even for low counts
  2802→        live_candles = []
  2803→        if asset.candles:
  2804→            hist_times = set(c['time'] for c in candles)
  2805→            for c in asset.candles:
  2806→                if c['time'] not in hist_times and c['time'] > (candles[-1]['time'] if candles else 0):
  2807→                    live_candles.append(c)
  2808→        asset.candles = candles + live_candles
  2809→        if asset.candles: asset.price = asset.candles[-1]['close']
  2810→        asset.write(force=True)
  2811→        # Update sync_state even for low counts
  2812→        if asset.candles:
  2813→            MT4_WRITER.sync_state[asset.symbol] = {
  2814→                'period': PERIOD,
  2815→                'candles': list(asset.candles),
  2816→                'last_write': 0,
  2817→                'last_price': float(asset.candles[-1]['close']) if asset.candles else 0.0,
  2818→            }
  2819→        # ===== FIX 3: exponential backoff retries =====
  2820→        if attempt < max_retries:
  2821→            delay = min(RETRY_BACKOFF_BASE * (2 ** (attempt - 1)), RETRY_BACKOFF_MAX)
  2822→            logmsg(f"[fetch_retry:{asset.symbol}] attempt {attempt}/{max_retries} got {len(candles)} candles; retry in {delay:.1f}s")
  2823→            await asyncio.sleep(delay)
  2824→    if last_error:
  2825→        logmsg(f"[fetch_retry:{asset.symbol}] all {max_retries} attempts failed. Last error: {last_error}")
  2826→    return len(candles), max_retries

  2835→def _subsecond_align(ts: int) -> int:
  2836→    """Align timestamp to subsecond window boundary (with TZ offset for consistency)."""
  2837→    tz = _get_tz_offset()
