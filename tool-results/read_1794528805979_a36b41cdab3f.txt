    86→async def wait_until(predicate: Callable[[], bool], *, timeout: float = 10.0,
    87→                      poll_interval: float = 0.05) -> None:
    88→    """ predicate() poll_interval True timeout.
    89→
    90→ asyncio.TimeoutError . :
    91→ while not condition:
    92→ await asyncio.sleep(X)
    93→ latency X poll_interval (50ms ).
    94→ """
    95→    async def _loop():
    96→        while not predicate():
    97→            await asyncio.sleep(poll_interval)
    98→    await asyncio.wait_for(_loop(), timeout=timeout)

   101→async def wait_for_first_event(*events: asyncio.Event, timeout: float = 10.0) -> int:
   102→    """ event index. TimeoutError .
   103→
   104→ :
   105→ for _ in range(100):
   106→ if cond_a: return A
   107→ if cond_b: return B
   108→ await asyncio.sleep(0.1)
   109→ """
   110→    if not events:
   111→        raise ValueError("wait_for_first_event requires at least one event")
   112→    tasks = [asyncio.ensure_future(e.wait()) for e in events]
   113→    try:
   114→        done, pending = await asyncio.wait(tasks, timeout=timeout,
   115→                                           return_when=asyncio.FIRST_COMPLETED)
   116→        for t in pending:
   117→            t.cancel()
   118→        for i, t in enumerate(tasks):
   119→            if t in done and t.result() and not t.cancelled():
   120→                return i
   121→        # timeout
   122→        raise asyncio.TimeoutError()
   123→    except asyncio.CancelledError:
   124→        for t in tasks:
   125→            if not t.done():
   126→                t.cancel()
   127→        raise

   130→def _schedule_event_set(event: Optional[asyncio.Event],
   131→                        loop: Optional[asyncio.AbstractEventLoop]) -> None:
   132→    """ asyncio.Event thread ( websocket thread).
   133→
   134→ run_coroutine_threadsafe coroutine 
   135→ callback loop thread-safe loop.call_soon_threadsafe.
   136→ """
   137→    if event is None or loop is None:
   138→        return
   139→    try:
   140→        if not loop.is_closed() and loop.is_running():
   141→            loop.call_soon_threadsafe(event.set)
   142→    except RuntimeError:
   143→        # loop —
   144→        pass

   160→def load_session(email: str, user_agent: str = None) -> dict:
   161→    if user_agent is None:
   162→        try: user_agent = UserAgent().random
   163→        except Exception: user_agent = USER_AGENT
   164→    output_file = Path(resource_path("session.json"))
   165→    with session_lock:
   166→        all_sessions = {}
   167→        if output_file.exists():
   168→            try: all_sessions = json.loads(output_file.read_text())
   169→            except json.JSONDecodeError: pass
   170→        else:
   171→            output_file.parent.mkdir(exist_ok=True, parents=True)
   172→        if email not in all_sessions:
   173→            all_sessions[email] = {"cookies": None, "token": None, "user_agent": user_agent}
   174→        output_file.write_text(json.dumps(all_sessions, indent=4))
   175→        return all_sessions.get(email)

   200→    __attrs__ = ['ssl_context', 'max_retries', 'config', '_pool_connections', '_pool_maxsize', '_pool_block', 'source_address']
   201→    def __init__(self, *args, **kwargs):
   202→        self.ssl_context = kwargs.pop('ssl_context', None)
   203→        self.cipherSuite = kwargs.pop('cipherSuite', 'ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384')
   204→        self.source_address = kwargs.pop('source_address', None)
   205→        self.server_hostname = kwargs.pop('server_hostname', None)
   206→        self.ecdhCurve = kwargs.pop('ecdhCurve', 'prime256v1')
   207→        if not self.ssl_context:
   208→            self.ssl_context = ssl.create_default_context(ssl.Purpose.SERVER_AUTH)
   209→            self.ssl_context.orig_wrap_socket = self.ssl_context.wrap_socket
   210→            self.ssl_context.wrap_socket = self.wrap_socket
   211→        if self.server_hostname:
   212→            self.ssl_context.server_hostname = self.server_hostname
   213→        if self.cipherSuite:
   214→            self.ssl_context.set_ciphers(self.cipherSuite)
   215→            self.ssl_context.set_ecdh_curve(self.ecdhCurve)
   216→            self.ssl_context.minimum_version = ssl.TLSVersion.TLSv1_2
   217→            self.ssl_context.maximum_version = ssl.TLSVersion.TLSv1_3
   218→        super().__init__(**kwargs)

   220→    def wrap_socket(self, *args, **kwargs):
   221→        if hasattr(self.ssl_context, 'server_hostname') and self.ssl_context.server_hostname:
   222→            kwargs['server_hostname'] = self.ssl_context.server_hostname
   223→            self.ssl_context.check_hostname = False
   224→        else:
   225→            self.ssl_context.check_hostname = True
   226→        return self.ssl_context.orig_wrap_socket(*args, **kwargs)

   228→    def init_poolmanager(self, *args, **kwargs):
   229→        kwargs['ssl_context'] = self.ssl_context
   230→        kwargs['source_address'] = self.source_address

   642→class EventRegistry:
   643→    def __init__(self):
   644→        self._events: Dict[str, asyncio.Event] = {}
   645→        self._data: Dict[str, Any] = {}
   646→        self._lock = asyncio.Lock()
   647→
   648→    async def get_event(self, key: str) -> asyncio.Event:
   649→        async with self._lock:
   650→            if key not in self._events:
   651→                self._events[key] = asyncio.Event()
   652→            return self._events[key]
   653→
   654→    async def set_event(self, key: str, data: Any = None):
   655→        async with self._lock:
   656→            if key not in self._events:
   657→                self._events[key] = asyncio.Event()
   658→            self._data[key] = data
   659→            self._events[key].set()
   660→
   661→    async def wait_event(self, key: str, timeout: float = 30.0) -> Any:
   662→        event = await self.get_event(key)
   663→        try:
   664→            await asyncio.wait_for(event.wait(), timeout=timeout)
   665→            return self._data.get(key)
   666→        except asyncio.TimeoutError:
   667→            return None
   668→
   669→    async def clear_event(self, key: str):
   670→        async with self._lock:
   671→            if key in self._events:
   672→                self._events[key].clear()
   673→            if key in self._data:
   674→                del self._data[key]

   941→def group_by_period(data, period):
   942→    grouped = defaultdict(list)
   943→    for tick in data:
   944→        timestamp = int(tick[0])
   945→        timeframe = int(timestamp // period)

   949→def calculate_candles(history, period):
   950→    if not isinstance(history, list) or not history: return []
   951→    grouped = group_by_period(history, period)
   952→    candles = []
   953→    for minute, ticks in grouped.items():
   954→        open_price = ticks[0][1]
   955→        close_price = ticks[-1][1]
   956→        high_price = max(tick[1] for tick in ticks)
   957→        low_price = min(tick[1] for tick in ticks)
   958→        candle = {'time': minute * period, 'open': open_price, 'close': close_price, 'high': high_price, 'low': low_price, 'ticks': len(ticks)}
   959→        candles.append(candle)
   960→    return candles[:-1] if len(candles) > 1 else candles

   971→def _parse_raw_candles(raw_candles):
   972→    """ WebSocket dicts OHLC .
   973→
   974→ "history/load":
   975→ 1) [time, open, close, high, low] — Quotex (5 )
   976→ 2) [time, open, close, high, low, vol] — volume (6 )
   977→ 3) {"time", "open", "high", "low", "close", ...} — dict
   978→ 4) [time, price(, volume)] — tick ( )
   979→ """
   980→    if not raw_candles: return []
   981→    parsed = []
   982→    for c in raw_candles:
   983→        try:
   984→            if isinstance(c, dict) and 'time' in c:
   985→                t = int(c.get('time', c.get('timestamp', 0)))
   986→                o = float(c.get('open', 0) or 0)
   987→                h = float(c.get('high', c.get('max', 0)) or 0)
   988→                l = float(c.get('low', c.get('min', 0)) or 0)
   989→                cl = float(c.get('close', c.get('c', 0)) or 0)
   990→                v = int(c.get('volume', c.get('vol', 0)) or 0)
   991→                if t > 0 and o > 0 and h > 0 and l > 0 and cl > 0:
   992→                    parsed.append({'time': t, 'open': o, 'high': h, 'low': l, 'close': cl, 'volume': v})
   993→            elif isinstance(c, (list, tuple)) and len(c) >= 5:
   994→                # Quotex WS: [time, open, close, high, low(, volume)]
   995→                t = int(c[0])
